Returns counts grouped by expiry status.

Items with a NULL expiry date are counted as good. Items expiring within expiringSoonDays days are counted as expiring. Items with a date before today are counted as expired.

Future<Map<String, int>> expiryDistribution(
  Database db, {
  required int inventoryId,
  required int expiringSoonDays,
}) async {
  final rows = await db.rawQuery(
    '''
    SELECT
      CASE
        WHEN expiry_date IS NULL THEN 'good'
        WHEN date(expiry_date) < date('now') THEN 'expired'
        WHEN date(expiry_date) <= date('now', '+' || ? || ' days')
          THEN 'expiring'
        ELSE 'good'
      END as status,
      COUNT(*) as cnt
    FROM inventory
    WHERE inventory_id = ?
    GROUP BY status
  ''',
    [expiringSoonDays, inventoryId],
  );
  return {
    for (final r in rows)
      r['status'] as String? ?? 'good': r['cnt'] as int? ?? 0,
  };
}
